```typescript
export function createTTSProvider(): TTSProvider {
  const raw = process.env.TTS_PROVIDER;
  const provider = (raw && raw.trim().length > 0 ? raw : "elevenlabs").toLowerCase();
  if (provider === "local") { return new LocalTTSProvider(); }
  if (provider === "elevenlabs") {
    if (!process.env.ELEVENLABS_API_KEY) { console.warn(...); return new LocalTTSProvider(); }
    return new FallbackTTSProvider(new ElevenLabsTTSProvider(), new LocalTTSProvider());
  }
  console.warn(`[voice] Unknown TTS_PROVIDER="${provider}", falling back to local`);
  return new LocalTTSProvider();
}
```
