**Step 1: Create `apps/desktop-py/tools/clone_voice.py`**

```python
# /// script
# requires-python = ">=3.10"
# dependencies = [
#   "kokoro>=0.9.4",
#   "kokoclone>=0.1.0",
#   "librosa>=0.10.0",
#   "torch>=2.0.0",
#   "soundfile>=0.12.0",
#   "rich>=13.0",
#   "pydantic>=2.7.0",
#   "python-dotenv>=1.0.0",
# ]
# ///
"""Clone a voice from reference audio file and save to JARVIS voice profile.

Standalone PEP 723 script — runs without full desktop-py environment.
Uses KokoClone ECAPA-TDNN encoder to extract speaker embedding from 3–30 seconds
of reference audio, saved as ~/.jarvis/voices/cloned_voice.pt.

Usage:
  uv run apps/desktop-py/tools/clone_voice.py reference.wav
  uv run apps/desktop-py/tools/clone_voice.py reference.mp3 --output /custom/path

Requirements:
  uv 0.4+ (PEP 723 inline deps support)
  Python 3.10+
  Reference audio: .wav, .mp3, .m4a, .ogg (3–30 seconds recommended)

After cloning, activate the voice in JARVIS:
  1. Run JARVIS: jd
  2. Type /config
  3. Select "Voz clonada" → enter the path shown after cloning
"""
import argparse
import sys
from pathlib import Path

# ---------------------------------------------------------------------------
# Paths
# ---------------------------------------------------------------------------
_JARVIS_DIR = Path.home() / ".jarvis"
_VOICES_DIR = _JARVIS_DIR / "voices"
_DEFAULT_OUTPUT = str(_VOICES_DIR / "cloned_voice.pt")


def _setup_console():
    """Initialize rich console for script output."""
    from rich.console import Console
    return Console()


def clone_voice(reference_audio: str, output_path: str = _DEFAULT_OUTPUT) -> str:
    """Clone voice from reference audio file.

    Validates audio duration, extracts ECAPA-TDNN speaker embedding,
    and saves .pt profile to output_path.

    Args:
        reference_audio: Path to .wav, .mp3, .m4a, etc. (3–30s recommended)
        output_path: Destination .pt file (default: ~/.jarvis/voices/cloned_voice.pt)

    Returns:
        str: Absolute path to saved cloned_voice.pt

    Raises:
        SystemExit: On file not found or encoder failure
    """
    console = _setup_console()

    # Validate file exists
    ref_path = Path(reference_audio)
    if not ref_path.exists():
        console.print(f"[red]ERRO: Arquivo não encontrado: {reference_audio}[/red]")
        sys.exit(1)

    # Lazy imports — heavy deps
    from jarvis_desktop.voice_cloning import (
        validate_reference_audio,
        extract_speaker_embedding,
        save_cloned_voice,
    )

    # Validate duration (warn but continue — D-02: user controls)
    console.print(f"[cyan]Verificando áudio: {reference_audio}[/cyan]")
    is_long_enough = validate_reference_audio(reference_audio, min_duration_sec=3.0)
    if not is_long_enough:
        console.print("[yellow]Aviso: duração mínima recomendada é 3 segundos. Continuando...[/yellow]")

    # Extract speaker embedding
    console.print("[cyan]Extraindo perfil de voz (ECAPA-TDNN)...[/cyan]")
    console.print("[dim]Isso pode levar 5–30 segundos na primeira execução (download de modelo).[/dim]")
    try:
        embedding = extract_speaker_embedding(reference_audio, device="cpu")
    except Exception as exc:
        console.print(f"[red]ERRO ao extrair embedding: {exc}[/red]")
        sys.exit(1)

    # Save profile
    output_dir = Path(output_path).parent
    output_name = Path(output_path).name
    console.print(f"[cyan]Salvando perfil em: {output_path}[/cyan]")

    # Override save dir if custom output specified
    import importlib.util
    import jarvis_desktop.voice_cloning as vc_module
    original_voices_dir = vc_module.VOICES_DIR
    if str(output_dir) != str(vc_module.VOICES_DIR):
        vc_module.VOICES_DIR = output_dir
    try:
        saved = save_cloned_voice(embedding, name=output_name)
    finally:
        vc_module.VOICES_DIR = original_voices_dir

    console.print(f"[green]Voz clonada com sucesso![/green]")
    console.print(f"[green]Perfil salvo em: {saved}[/green]")
    console.print()
    console.print("[dim]Para ativar no JARVIS:[/dim]")
    console.print(f"[dim]  1. Execute: jd[/dim]")
    console.print(f"[dim]  2. Digite: /config[/dim]")
    console.print(f"[dim]  3. Selecione 'Voz clonada' e informe: {saved}[/dim]")

    return saved


if __name__ == "__main__":
    parser = argparse.ArgumentParser(
        description="Clona voz de arquivo de áudio de referência para uso no JARVIS.",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Exemplos:
  uv run apps/desktop-py/tools/clone_voice.py minha_voz.wav
  uv run apps/desktop-py/tools/clone_voice.py gravacao.mp3 --output ~/minha_voz.pt

Requisitos de qualidade:
  - Mínimo: 3 segundos
  - Recomendado: 5–10 segundos com fala natural
  - Ideal: 10–30 segundos com variação de entonação
  - Formatos: .wav, .mp3, .m4a, .ogg
        """,
    )
    parser.add_argument(
        "reference",
        help="Caminho para o arquivo de áudio de referência (.wav, .mp3, etc.)",
    )
    parser.add_argument(
        "--output",
        default=_DEFAULT_OUTPUT,
        help=f"Caminho de saída para o perfil .pt (padrão: {_DEFAULT_OUTPUT})",
    )
    args = parser.parse_args()
    clone_voice(args.reference, args.output)
```

```python
def _menu_cloned_voice(config: JarvisConfig) -> None:
    """Config menu: activate or deactivate cloned voice (Phase 85, D-04).

    Shows current state, lets user enter path to .pt file or clear to disable.
    """
    from jarvis_desktop.config import save_config  # Local import avoids circular
    console = _console()

    current = config.cloned_voice_path or "(desativada)"
    console.print(f"Voz clonada atual: {current}")
    console.print()
    console.print("Opcoes:")
    console.print("  - Digite o caminho para o arquivo .pt (ex: ~/.jarvis/voices/cloned_voice.pt)")
    console.print("  - Digite 'desativar' para usar a voz padrao (kokoro_voice)")
    console.print("  - Pressione Enter para manter o estado atual")
    console.print()
    choice = input("Caminho ou 'desativar': ").strip()

    if not choice:
        console.print("Nenhuma alteracao.")
        return

    if choice.lower() in ("desativar", "disable", "off", "none", "0"):
        config.cloned_voice_path = ""
        save_config(config)
        console.print("Voz clonada desativada. Usando voz padrao.")
        return

    # Expand ~ and validate path existence
    from pathlib import Path
    expanded = str(Path(choice).expanduser())
    if not Path(expanded).exists():
        console.print(f"[VOICE] Arquivo nao encontrado: {expanded}")
        console.print("Caminho nao alterado.")
        return

    config.cloned_voice_path = expanded
    save_config(config)
    console.print(f"Voz clonada ativada: {expanded}")
```

Use ASCII characters only in user-facing text (no Unicode box-drawing) — same pattern as existing menu items in chat.py to avoid UnicodeEncodeError on Windows cp1252.
