<!-- Algorithm from CONTEXT.md D-10 (exact code to implement) -->
```typescript
function isInQuietHours(now: Date, startHHMM: string, endHHMM: string): boolean {
  const [startH, startM] = startHHMM.split(':').map(Number);
  const [endH, endM] = endHHMM.split(':').map(Number);
  const startMin = startH * 60 + startM;
  const endMin = endH * 60 + endM;
  const nowMin = now.getHours() * 60 + now.getMinutes();
  return startMin > endMin
    ? (nowMin >= startMin || nowMin < endMin)   // cross-midnight
    : (nowMin >= startMin && nowMin < endMin);  // same-day
}
```
