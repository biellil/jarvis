    import type { ProactiveEvent } from '../../shared/ipc-types';

    /** Phase 67 (D-09, D-14): builds pt-BR TTS string for each proactive event kind */
    export function buildProactiveTtsText(evt: ProactiveEvent): string {
      switch (evt.kind) {
        case 'reminder':
          return `Lembrete: ${evt.message}`;
        case 'folder_event': {
          const folderName = evt.folderPath.split('/').pop() ?? evt.folderPath;
          if (evt.files.length === 1) {
            return `Novo arquivo em ${folderName}: ${evt.files[0]!.name}`;
          }
          if (evt.files.length <= 5) {
            return `Chegaram ${evt.files.length} arquivos em ${folderName}: ${evt.files.map((f) => f.name).join(', ')}`;
          }
          const first3 = evt.files.slice(0, 3).map((f) => f.name).join(', ');
          return `Chegaram ${evt.files.length} arquivos em ${folderName}: ${first3} e mais ${evt.files.length - 3}`;
        }
        case 'daily_summary':
          return evt.text;
        default:
          return 'Nova notificação proativa';
      }
    }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Renderer → IPC applyFolderWatch | folderWatch.path from user input — validated in IPC handler (Plan 67-08) before reaching backend |
| ProactiveEvent → chat bubble render | Event from backend SSE via main process IPC; content rendered as text only, no dangerouslySetInnerHTML |
| 'proactive:event' IPC channel → renderer | Main process is trusted sender; renderer consumes without additional validation (single-user desktop, no cross-origin risk) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-67-03 | Tampering | ProactiveSection folder path input | mitigate | Renderer shows Field.Error for paths that don't exist (validates via IPC before applying). IPC handler in Plan 67-08 performs isAbsolute + DENIED_PATHS check. Double validation in renderer + main. |
| T-67-05 | Injection | ProactiveEventBubble content rendering | accept | Content is rendered via React JSX (`{event.message}`) — React's JSX auto-escapes strings, preventing XSS. No dangerouslySetInnerHTML used. LLM-generated daily_summary text is rendered as plain text. |
| T-67-06 | Injection | buildProactiveTtsText output → speakText | accept | TTS input is plain string assembled from known fields (message, file names, summary text). No shell execution. kokoro/Murf synthesize speech from plain text — no command injection surface. |
</threat_model>
