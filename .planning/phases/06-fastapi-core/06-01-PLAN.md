**src/jarvis/api/lifespan.py** — Lifespan context manager per D-01 (sessao unica global):
```python
import os
from contextlib import asynccontextmanager
from fastapi import FastAPI
from loguru import logger
from jarvis.config import settings
from jarvis.llm.factory import create_llm
from jarvis.memory.store import MemoryStore
from jarvis.memory.vectors import MemoryVectors
from jarvis.core.session import ChatSession
from jarvis.tools import ALL_TOOLS
from jarvis.executor import ActionExecutor

@asynccontextmanager
async def lifespan(app: FastAPI):
    logger.info("JARVIS API starting up...")
    os.makedirs(os.path.dirname(settings.sqlite_path) or ".", exist_ok=True)
    os.makedirs(settings.chroma_path, exist_ok=True)
    llm = create_llm()
    db = MemoryStore(settings.sqlite_path)
    vectors = MemoryVectors(settings.chroma_path)
    executor = ActionExecutor()
    session = ChatSession(llm=llm, db=db, vectors=vectors, tools=ALL_TOOLS, executor=executor)
    app.state.session = session
    app.state.db = db
    app.state.vectors = vectors
    yield
    await session.save()
    db.close()
    logger.info("JARVIS API shut down.")
```

**src/jarvis/api/routes/health.py** — Health endpoints (API-03, API-04):
```python
from pathlib import Path
from fastapi import APIRouter, Request, HTTPException
from jarvis.config import settings

router = APIRouter()

@router.get("/health")
async def health():
    return {"status": "ok"}

@router.get("/health/ready")
async def health_ready(request: Request):
    errors = []
    # Check SQLite file exists
    if not Path(settings.sqlite_path).exists():
        errors.append("sqlite: file not found")
    # Check ChromaDB via existing client (Pitfall 4: never create new PersistentClient per request)
    try:
        vectors = request.app.state.vectors
        vectors._client.heartbeat()
    except Exception as e:
        errors.append(f"chromadb: {e}")
    if errors:
        raise HTTPException(status_code=503, detail={"status": "not_ready", "errors": errors})
    return {"status": "ok", "stores": {"sqlite": "ok", "chromadb": "ok"}}
```
