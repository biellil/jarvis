```typescript
// apps/backend-ts/src/index.ts (or gateway.ts — wherever ChatSession.create() is called)
import type { CaptureScreenFn } from './session/vision-tool.js';

const captureScreenFn: CaptureScreenFn = async () => {
  // Call Electron CAPTURE_SCREEN via the gateway HTTP back-channel
  // Pattern: same as dispatch-action — call Electron via the clientId bridge
  // For Phase 63 MVP: use a dedicated /internal/capture-screen endpoint
  // that the Electron main process registers via express or IPC bridge
  try {
    const gatewayUrl = process.env['GATEWAY_URL'] ?? 'http://localhost:3000';
    const response = await fetch(
      `${gatewayUrl.replace(/^ws:\/\//i, 'http://').replace(/^wss:\/\//i, 'https://')}/internal/capture-screen`,
      { method: 'POST', signal: AbortSignal.timeout(5_000) },
    );
    if (!response.ok) return { success: false, error: `HTTP ${response.status}` };
    return await response.json() as { success: true; base64: string } | { success: false; error: string };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
};
```
