Create `Dockerfile.python` at repo root with two stages (per D-03, D-04):

**Stage 1 — builder (FROM python:3.12-slim AS builder):**
```dockerfile
FROM python:3.12-slim AS builder

WORKDIR /app

# Layer cache: copy only dep manifest first
COPY pyproject.toml ./

# Install all Python deps (includes build-time native compilation)
RUN pip install --no-cache-dir --upgrade pip && \
    pip install --no-cache-dir .

# Copy application source
COPY src/ ./src/

# Install the package itself (so jarvis is importable)
RUN pip install --no-cache-dir -e .
```

**Stage 2 — runtime (FROM python:3.12-slim AS runtime):**
```dockerfile
FROM python:3.12-slim AS runtime

WORKDIR /app

# CRITICAL: system libs needed by native Python packages at runtime
# libgomp1 — ctranslate2/faster-whisper (OpenMP)
# libsndfile1 — soundfile/kokoro
# libportaudio2 — sounddevice
# espeak-ng — kokoro phonemizer on Linux
# curl — healthcheck CMD
RUN apt-get update && apt-get install -y --no-install-recommends \
    libgomp1 \
    libsndfile1 \
    libportaudio2 \
    espeak-ng \
    curl \
    && rm -rf /var/lib/apt/lists/*

# Copy installed Python packages from builder
COPY --from=builder /usr/local/lib/python3.12/site-packages \
                    /usr/local/lib/python3.12/site-packages
COPY --from=builder /usr/local/bin /usr/local/bin

# Copy application source
COPY --from=builder /app/src ./src
COPY --from=builder /app/pyproject.toml ./

# Ensure data directory exists (will be overridden by volume mount)
RUN mkdir -p /app/data

# Single worker — in-memory session_store breaks with >1 (per D-04, Phase 6 D-01)
CMD ["uvicorn", "jarvis.api:app", "--host", "0.0.0.0", "--port", "8000", "--workers", "1"]
```

Create `Dockerfile.node` at repo root with two stages (per D-02). Build context is repo root so we can access `apps/gateway/`.

**Stage 1 — builder (FROM node:22-slim AS builder):**
```dockerfile
FROM node:22-slim AS builder

WORKDIR /app

# Copy only dep manifest for layer cache
COPY apps/gateway/package.json ./

# Use npm (not pnpm) inside container for simplicity (per D-02 discussion)
# No package-lock.json exists — npm install resolves from package.json
RUN npm install

# Copy TypeScript source and config, then compile
COPY apps/gateway/src ./src
COPY apps/gateway/tsconfig.json ./
RUN npx tsc
```

**Stage 2 — runtime (FROM node:22-slim AS runtime):**
```dockerfile
FROM node:22-slim AS runtime

WORKDIR /app

# Copy package.json and install production deps only
COPY --from=builder /app/package.json ./
RUN npm install --omit=dev

# Copy compiled JavaScript
COPY --from=builder /app/dist ./dist

# env vars injected by docker-compose env_file — NOT --env-file flag
CMD ["node", "dist/index.js"]
```

The `@tsconfig/node22` package is a devDependency used by tsconfig.json `extends`. It is needed in the builder stage during `npx tsc`. The `npm install` in builder installs ALL deps including dev, which includes `@tsconfig/node22` and `typescript`. The runtime stage `npm install --omit=dev` skips it correctly.
