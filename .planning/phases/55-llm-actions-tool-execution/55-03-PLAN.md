**Criar `apps/desktop/src/main/actions/open-folder.ts`:**
```typescript
import { shell } from 'electron';
import type { ActionHandler } from './types.js';
import { ok, fail } from './types.js';
import { describeError } from './validators.js';

export const openFolderHandler: ActionHandler = async (args) => {
  const path = args['path'];
  if (typeof path !== 'string' || path.length === 0) {
    return fail('invalid_args: path must be a non-empty string');
  }
  try {
    const errorMessage = await shell.openPath(path);
    if (errorMessage) {
      // shell.openPath returns '' on success, error string on failure
      return fail(`subprocess_failed: ${errorMessage}`);
    }
    return ok(`opened folder: ${path}`);
  } catch (err) {
    return fail(describeError(err));
  }
};
```

import type { ActionHandler } from './types.js';
import { ok, fail } from './types.js';
import { assertSafeAppName, runExecFile, describeError } from './validators.js';

export const closeFileHandler: ActionHandler = async (args) => {
  try {
    const processName = assertSafeAppName(args['processName']);
    if (process.platform === 'win32') {
      await runExecFile('taskkill', ['/IM', processName, '/F']);
    } else {
      await runExecFile('pkill', ['-f', processName]);
    }
    return ok(`process killed: ${processName}`);
  } catch (err) {
    return fail(describeError(err));
  }
};
```

import { ok, fail } from './types.js';

const MAX_FILE_SIZE = 1_048_576; // 1MB — per D-04

export const viewContentHandler: ActionHandler = async (args) => {
  const path = args['path'];
  if (typeof path !== 'string' || path.length === 0) {
    return fail('invalid_args: path must be a non-empty string');
  }
  try {
    const stats = await fs.stat(path);
    if (stats.size >= MAX_FILE_SIZE) {
      return fail(`file too large: ${stats.size} bytes > 1MB limit`);
    }
    const content = await fs.readFile(path, 'utf-8');

    console.debug(`[viewContent] read ${stats.size} bytes from path (first 100: ${content.slice(0, 100)})`);
    return ok(content);
  } catch (err) {
    return fail(describeError(err));
  }
};

import { ipcMain } from 'electron';
import { IPC_CHANNELS } from '../../shared/ipc-types.js';
import type { ActionAckPayload, ExecuteActionPayload } from '../../shared/ipc-types.js';
import { sendActionAck } from '../actions/actionsClient.js';
import { ACTION_HANDLERS } from '../actions/index.js';

export function setupActionsIpcHandlers(): void {
  // Existing: ACTION_ACK handler (Phase 54)
  ipcMain.handle(IPC_CHANNELS.ACTION_ACK, (_event, payload: ActionAckPayload) => {
    try {
      sendActionAck(payload.requestId, payload.status);
      return { success: true };
    } catch (err) {
      console.error('[actions-ipc] sendActionAck failed', err);
      return { success: false, error: (err as Error).message };
    }
  });

  // Phase 55: ACTION_EXECUTE — renderer confirmed action, execute in OS
  ipcMain.handle(IPC_CHANNELS.ACTION_EXECUTE, async (_event, payload: ExecuteActionPayload) => {
    const handler = ACTION_HANDLERS[payload.action];
    if (!handler) {
      console.error(`[actions-ipc] No handler for action: ${payload.action}`);
      return { ok: false, error: `unknown_action: ${payload.action}` };
    }
    try {
      // closeFile receives processName; all others receive path
      const args: Record<string, unknown> =
        payload.action === 'closeFile'
          ? { processName: payload.path }  // for closeFile, path field carries processName
          : { path: payload.path };
      const result = await handler(args);
      return {
        ok: result.success,
        content: result.success && result.output ? result.output : undefined,
        error: result.success ? undefined : result.error,
      };
    } catch (err) {
      console.error('[actions-ipc] Handler threw unexpectedly', err);
      return { ok: false, error: (err as Error).message };
    }
  });
}
```
