import { stat, readFile } from 'node:fs/promises';

  beforeEach(() => vi.resetAllMocks());

  it('calls shell.openPath with the file path and returns ok', async () => {

  });

  });
});

describe('viewContentHandler', () => {
  beforeEach(() => vi.resetAllMocks());

  it('reads file under 1MB and returns content', async () => {

    expect(result.error).toMatch(/path_not_found|not found/i);
  });
});
