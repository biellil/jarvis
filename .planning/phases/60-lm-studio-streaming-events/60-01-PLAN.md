LM Studio native SSE event format (official spec from planning context):
- event: chat.start  → data: { "type": "chat.start", "model_instance_id": string }
- event: message.delta → data: { "type": "message.delta", "content": string }  ← top-level content
- event: reasoning.delta → data: { "type": "reasoning.delta", "content": string }
- event: message.start → data: { "type": "message.start" }
- event: message.end  → data: { "type": "message.end" }
- event: chat.end → data: { "type": "chat.end", "result": { "model_instance_id": string, "output": array, "stats": { "input_tokens": number, "total_output_tokens": number, "tokens_per_second": number, "time_to_first_token_seconds": number }, "response_id": string } }
- event: error → data: { "type": "error", "error": { "type": string, "message": string, "code"?: string } }
- Native URL: replace /v1 with /api/v1 → endpoint: {base}/api/v1/chat
  Example: http://localhost:1234/v1 → http://localhost:1234/api/v1/chat

function makeSSEStream(events: Array<{ type: string; data: object }>): ReadableStream<Uint8Array> {
  const encoder = new TextEncoder();
  const chunks = events.map(e => encoder.encode(`event: ${e.type}\ndata: ${JSON.stringify(e.data)}\n\n`));
  let idx = 0;
  return new ReadableStream({
    pull(controller) {
      if (idx < chunks.length) controller.enqueue(chunks[idx++]);
      else controller.close();
    },
  });

GREEN PHASE: Create streaming-events.ts with ChatOpenAIStreamingEvents class:

```typescript
import { ChatOpenAI } from '@langchain/openai';
import type { BaseMessage } from '@langchain/core/messages';
import { AIMessageChunk } from '@langchain/core/messages';

export class ChatOpenAIStreamingEvents extends ChatOpenAI {
  private readonly nativeEventsEnabled: boolean;

  constructor(opts: ConstructorParameters<typeof ChatOpenAI>[0] & { nativeEventsEnabled: boolean }) {
    const { nativeEventsEnabled, ...rest } = opts;
    super(rest);
    this.nativeEventsEnabled = nativeEventsEnabled;
  }

  _buildNativeUrl(): string {
    const base: string = (this as any).clientConfig?.baseURL ?? 'http://localhost:1234/v1';
    // Replace /v1 suffix with /api/v1 (idempotent: if already /api/v1, this is a no-op)
    const withoutV1 = base.replace(/\/v1\/?$/, '');
    return `${withoutV1}/api/v1/chat`;
  }

  async *stream(input: BaseMessage[], ...args: unknown[]): AsyncGenerator<AIMessageChunk> {
    if (!this.nativeEventsEnabled) {
      yield* super.stream(input, ...args as Parameters<typeof super.stream>[1][]);
      return;
    }
    try {
      yield* this._streamNativeEvents(input);
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      console.warn('[ChatOpenAIStreamingEvents] Native events failed, falling back to standard SSE:', msg);
      yield* super.stream(input, ...args as Parameters<typeof super.stream>[1][]);
    }
  }

  async *_streamNativeEvents(messages: BaseMessage[]): AsyncGenerator<AIMessageChunk> {
    const chatMessages = messages.map(m => ({
      role: m._getType() === 'human' ? 'user' : (m._getType() === 'ai' ? 'assistant' : 'system'),
      content: typeof m.content === 'string' ? m.content : JSON.stringify(m.content),
    }));

    const url = this._buildNativeUrl();
    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ messages: chatMessages, stream: true }),
    });

    if (!response.ok) {
      throw new Error(`LM Studio native endpoint error: ${response.status} ${response.statusText}`);
    }

    const reader = response.body?.getReader();
    if (!reader) throw new Error('No response body reader available');

    let buffer = '';
    const decoder = new TextDecoder();

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });

      // Split by double newline (SSE event boundary)
      const parts = buffer.split('\n\n');
      buffer = parts.pop() ?? ''; // Keep incomplete event in buffer

      for (const part of parts) {
        const trimmed = part.trim();
        if (!trimmed) continue;

        const lines = trimmed.split('\n');
        const eventLine = lines.find(l => l.startsWith('event: '));
        const dataLine = lines.find(l => l.startsWith('data: '));
        if (!eventLine || !dataLine) continue;

        const eventType = eventLine.slice('event: '.length).trim();
        let data: Record<string, unknown>;
        try {
          data = JSON.parse(dataLine.slice('data: '.length).trim());
        } catch {
          continue; // Skip malformed JSON
        }

        switch (eventType) {
          case 'message.delta':
            // Official spec: { type: 'message.delta', content: string } — content is TOP-LEVEL
            if (typeof data.content === 'string' && data.content.length > 0) {
              yield new AIMessageChunk({ content: data.content, additional_kwargs: {} });
            }
            break;
          case 'reasoning.delta':
            // Log reasoning for observability (D-05: absorbed in backend, not forwarded to UI in Phase 60)
            console.log('[native-events] reasoning:', data.content);
            break;
          case 'chat.end': {
            const stats = (data.result as any)?.stats;
            if (stats?.time_to_first_token_seconds !== undefined) {
              console.log('[native-events] TTFT:', stats.time_to_first_token_seconds, 's, tps:', stats.tokens_per_second);
            }
            break;
          }
          case 'error': {
            const errData = data.error as { message?: string } | undefined;
            throw new Error(`LM Studio error event: ${errData?.message ?? JSON.stringify(data.error)}`);
          }
          // chat.start, message.start, message.end, prompt_processing.*, model_load.*, tool_call.* — ignored in Phase 60
          default:
            break;
        }
      }
    }
  }
}
```
