```python
def _make_input_feeder(inputs):
    """Returns a function that yields inputs in order, raising EOFError when exhausted."""
    it = iter(inputs)
    def feed(prompt=""):
        try:
            return next(it)
        except StopIteration:
            raise EOFError
    return feed
```
