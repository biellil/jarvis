    Criar `apps/desktop-py/tests/test_e2e_pipeline.py`:
    ```python
    """Teste E2E do pipeline PTT → STT → LLM(mock) → TTS (POL-04, D-13..D-18).

    Marker @pytest.mark.e2e — rodar isolado com `pytest -m e2e`.
    Unit suite ignora com `pytest -m "not e2e"`.
    """
    from __future__ import annotations

    import numpy as np
    import pytest


    @pytest.mark.e2e
    def test_ptt_full_pipeline_with_mocked_llm(
        tmp_home,
        e2e_audio_wav,
        monkeypatch,
        httpx_mock,
    ):
        """Pipeline PTT completo com STT real (Whisper tiny), LLM mockado, TTS real.

        Asserções (D-18):
          1. STT transcreveu (string não-vazia)
          2. HTTP request disparada com headers corretos
          3. SSE chunks parseados (tokens acumulados)
          4. TTS recebeu texto (sounddevice.play chamado)
          5. UI state sequence: listening → thinking → speaking → idle
        """
        # --- Mocks de hardware ---
        play_calls: list[tuple] = []
        ui_states: list[str] = []

        # Capturar de sounddevice — mock rec retorna fixture, mock play coleta chamadas
        def fake_rec(frames, **kwargs):
            class _FakeRecording:
                def __init__(self, arr):
                    self._arr = arr
                def __iter__(self):
                    return iter(self._arr)
            # sounddevice.rec retorna ndarray; ajustar conforme API real do JARVIS
            return e2e_audio_wav.reshape(-1, 1) if "channels" in kwargs else e2e_audio_wav

        monkeypatch.setattr("sounddevice.rec", fake_rec)
        monkeypatch.setattr("sounddevice.wait", lambda: None)
        monkeypatch.setattr(
            "sounddevice.play",
            lambda data, samplerate=None, **kw: play_calls.append((np.asarray(data).copy(), samplerate)),
        )
        # Se record_until_silence usa sounddevice.InputStream, mockar também:
        monkeypatch.setattr(
            "jarvis_desktop.stt.record_until_silence",
            lambda **kw: e2e_audio_wav,
            raising=False,
        )

        # UI state spy — patch ui.set_state se existir, senão coletar via outro hook
        try:
            monkeypatch.setattr(
                "jarvis_desktop.ui.set_state",
                lambda s: ui_states.append(s),
                raising=True,
            )
        except AttributeError:
            # Se não houver set_state, capturar via print/console; fallback
            pass

        # --- Mock LLM gateway via pytest-httpx ---
        from jarvis_desktop.config import JarvisConfig
        config = JarvisConfig(whisper_model="tiny", tts_provider="kokoro")
        gateway_url = getattr(config, "gateway_url", "http://localhost:3001/chat")

        sse_body = (
            b"event: token\ndata: ola\n\n"
            b"event: token\ndata:  humano\n\n"
            b"event: end\ndata: \n\n"
        )
        httpx_mock.add_response(
            method="POST",
            url=gateway_url,
            content=sse_body,
            headers={"Content-Type": "text/event-stream"},
        )

        # --- Disparar 1 ciclo PTT ---
        # IMPORTANT: o executor deve identificar a função pública correta para 1-shot PTT.
        # Possíveis pontos de entrada:
        #   - jarvis_desktop.voice_modes.run_ptt_mode(config, max_turns=1)
        #   - jarvis_desktop.chat.chat_loop(config) com config.voice_mode="ptt" e queue de stop
        # Verificar voice_modes.py e usar a API existente. Se necessário, refatorar p/ expor
        # uma helper `run_single_ptt_turn(config, audio)` (mantém o teste estável).

        from jarvis_desktop import voice_modes
        # Exemplo (ajustar para API real):
        result = voice_modes.run_ptt_once(config) if hasattr(voice_modes, "run_ptt_once") else None
        if result is None:
            pytest.skip(
                "voice_modes não expõe API single-turn. Refatorar voice_modes.run_ptt_mode "
                "para extrair `run_ptt_once(config)` antes de habilitar este teste."
            )

        # --- Asserções (D-18) ---

        # 1. STT transcreveu
        transcript = result.get("transcript", "")
        assert isinstance(transcript, str) and len(transcript.strip()) > 0, (
            f"STT deve retornar string não-vazia, recebeu: {transcript!r}"
        )
        assert any(len(w) >= 2 for w in transcript.split()), (
            f"STT deve produzir pelo menos uma palavra reconhecível, recebeu: {transcript!r}"
        )

        # 2. HTTP request disparada com headers corretos
        requests = httpx_mock.get_requests()
        assert len(requests) >= 1, "Pelo menos 1 request HTTP para o gateway"
        req = requests[0]
        assert req.headers.get("Content-Type", "").startswith("application/json") or \
               "json" in req.headers.get("content-type", "").lower(), \
               f"Content-Type esperado application/json, got {req.headers.get('Content-Type')}"

        # 3. SSE chunks parseados — tokens acumulados
        full_response = result.get("response", "")
        assert "ola" in full_response.lower() or "humano" in full_response.lower(), (
            f"Tokens SSE devem ser acumulados, response={full_response!r}"
        )

        # 4. TTS recebeu texto — sounddevice.play foi chamado
        assert len(play_calls) >= 1, "sounddevice.play deve ter sido chamado pelo menos 1×"
        played_audio, sr = play_calls[0]
        assert isinstance(played_audio, np.ndarray) and played_audio.size > 0, (
            "TTS deve produzir áudio não-vazio"
        )

        # 5. UI state sequence (D-18.5)
        if ui_states:
            expected_states = ["listening", "thinking", "speaking"]
            for state in expected_states:
                assert state in ui_states, (
                    f"Estado '{state}' esperado em sequence, got {ui_states}"
                )
    ```
