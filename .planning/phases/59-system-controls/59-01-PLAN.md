**Step 2 — Create apps/desktop/src/main/actions/adjust-volume.ts:**

Handler reads current volume, applies delta, clamps to [0, 100], sets new volume. Each platform uses different CLI. Use `Math.max(0, Math.min(100, current + delta))` for clamping.

Platform commands:
- **Linux:** `pactl get-sink-volume @DEFAULT_SINK@` → parse first `%` value (e.g. "50%") from stdout. Then `pactl set-sink-volume @DEFAULT_SINK@ <newLevel>%`
- **macOS:** `osascript -e "output volume of (get volume settings)"` → returns integer. Then `osascript -e "set volume output volume <newLevel>"`
- **Windows:** PowerShell one-liner to get volume: `powershell.exe -NoProfile -Command "(Get-AudioDevice -Playback).Volume"` (returns float 0-100). Then set: `powershell.exe -NoProfile -Command "Set-Volume -Level <newLevel>"`. If `Set-Volume` not found (ENOENT or subprocess_failed containing 'not recognized'), try fallback: `nircmd.exe setsysvolume <nircmdLevel>` where nircmdLevel = Math.round(newLevel / 100 * 65535). If nircmd also fails, return fail('Set-Volume cmdlet not available and nircmd not found — install nircmd or update Windows audio modules').

**Step 3 — Create apps/desktop/src/main/actions/toggle-mute.ts:**

Platform commands:
- **Linux:** `pactl get-sink-mute @DEFAULT_SINK@` → stdout is "Mute: yes" or "Mute: no". Then `pactl set-sink-mute @DEFAULT_SINK@ toggle`
- **macOS:** `osascript -e "output muted of (get volume settings)"` → "true" or "false". Then `osascript -e "set volume output muted <true|false>"`
- **Windows:** `powershell.exe -NoProfile -Command "(Get-AudioDevice -Playback).Mute"` or via nircmd. Use `nircmd.exe mutesysvolume 2` (toggle = 2 in nircmd) as the simplest cross-version approach. For getting state, try PowerShell first.

```typescript
import type { ActionHandler } from './types.js';
import { ok, fail } from './types.js';
import { describeError, runExecFile } from './validators.js';

export const toggleMuteHandler: ActionHandler = async (args) => {
  void args; // no args needed
  try {
    let newState: 'muted' | 'unmuted';
    if (process.platform === 'win32') {
      newState = await toggleMuteWindows();
    } else if (process.platform === 'darwin') {
      newState = await toggleMuteMac();
    } else {
      newState = await toggleMuteLinux();
    }
    return ok(newState === 'muted' ? 'muted' : 'unmuted');
  } catch (err) {
    return fail(describeError(err));
  }
};

async function toggleMuteLinux(): Promise<'muted' | 'unmuted'> {
  const { stdout } = await runExecFile('pactl', ['get-sink-mute', '@DEFAULT_SINK@']);
  const wasMuted = stdout.toLowerCase().includes('yes');
  await runExecFile('pactl', ['set-sink-mute', '@DEFAULT_SINK@', 'toggle']);
  return wasMuted ? 'unmuted' : 'muted';
}

async function toggleMuteMac(): Promise<'muted' | 'unmuted'> {
  const { stdout } = await runExecFile('osascript', ['-e', 'output muted of (get volume settings)']);
  const wasMuted = stdout.trim().toLowerCase() === 'true';
  const newMuted = !wasMuted;
  await runExecFile('osascript', ['-e', `set volume output muted ${newMuted}`]);
  return newMuted ? 'muted' : 'unmuted';
}

async function toggleMuteWindows(): Promise<'muted' | 'unmuted'> {
  // Try PowerShell to get mute state
  let wasMuted = false;
  try {
    const { stdout } = await runExecFile('powershell.exe', [
      '-NoProfile', '-Command',
      '[bool](Get-AudioDevice -Playback | Select-Object -ExpandProperty Mute)',
    ]);
    wasMuted = stdout.trim().toLowerCase() === 'true';
  } catch { /* assume unmuted */ }

  // Toggle: try nircmd first (works without modules), then PowerShell
  try {
    await runExecFile('nircmd.exe', ['mutesysvolume', '2']); // 2 = toggle
    return wasMuted ? 'unmuted' : 'muted';
  } catch {
    try {
      await runExecFile('powershell.exe', [
        '-NoProfile', '-Command',
        `Set-AudioDevice -Playback -Mute ${!wasMuted}`,
      ]);
      return wasMuted ? 'unmuted' : 'muted';
    } catch {
      throw new Error('Unable to toggle mute — install nircmd or AudioDeviceCmdlets PowerShell module');
    }
  }
}
```

**Step 1 — Create apps/desktop/src/main/actions/media-control.ts:**

Platform commands:
- **Linux:** `playerctl play-pause` / `playerctl next` / `playerctl previous`. If playerctl ENOENT, the ActionValidationError('command_not_found', 'playerctl') from runExecFile maps to fail('command_not_found: playerctl — install playerctl to control media')
- **macOS:** `osascript -e "tell application \"System Events\" to key code 16 using {}"` — but media key codes differ. Use these AppleScript key codes: play/pause = 100 (F8 key-equivalent using key code 100 with no modifiers via System Events). Actually use the correct approach: `osascript -e "tell application \"System Events\" to keystroke \" \" using {command down}"` won't work. Use `osascript` with key codes: play_pause=16 doesn't work reliably. Best macOS approach: use `osascript -e "tell application \"System Events\" to key code 49"` does space. For actual media keys use NX key codes via osascript:
  ```
  play_pause: osascript -e "tell application \"System Events\" to key code 100"  -- F8 / play-pause
  next_track:  osascript -e "tell application \"System Events\" to key code 101"  -- F9 / next
  prev_track:  osascript -e "tell application \"System Events\" to key code 98"   -- F7 / prev
  ```
  These NX media key codes work via System Events (requires Accessibility permission).
- **Windows:** `powershell.exe -NoProfile -Command "(New-Object -ComObject WScript.Shell).SendKeys([char]179)"` where char codes: play_pause=179, next=176, prev=177. Or use nircmd: `nircmd.exe sendkeypress media_play_pause` / `nircmd.exe sendkeypress media_next_track` / `nircmd.exe sendkeypress media_prev_track`

**VALID_COMMANDS** set: `['play_pause', 'next_track', 'prev_track']` — validate before dispatch, throw ActionValidationError if unknown.

**macOS Accessibility check:** Before running osascript for media keys, check if accessibility is available. Use Electron's `systemPreferences.getMediaAccessStatus` — but that's not available for Accessibility. Instead, attempt the osascript command and catch `EACCES` or stderr containing "not allowed". If caught, return fail('media_control requires Accessibility permission — grant JARVIS access in System Settings → Privacy & Security → Accessibility').

```typescript
import type { ActionHandler } from './types.js';
import { ok, fail } from './types.js';
import { ActionValidationError, describeError, runExecFile } from './validators.js';

const VALID_COMMANDS = new Set(['play_pause', 'next_track', 'prev_track']);

export const mediaControlHandler: ActionHandler = async (args) => {
  try {
    const command = args['command'];
    if (typeof command !== 'string' || !VALID_COMMANDS.has(command)) {
      throw new ActionValidationError('invalid_args', `command must be one of: ${[...VALID_COMMANDS].join(', ')}`);
    }

    if (process.platform === 'win32') {
      await mediaControlWindows(command);
    } else if (process.platform === 'darwin') {
      await mediaControlMac(command);
    } else {
      await mediaControlLinux(command);
    }
    return ok(`media: ${command}`);
  } catch (err) {
    if (err instanceof ActionValidationError && err.code === 'command_not_found') {
      return fail(`${err.message} — install playerctl to control media on Linux`);
    }
    return fail(describeError(err));
  }
};

const LINUX_COMMANDS: Record<string, string> = {
  play_pause: 'play-pause',
  next_track: 'next',
  prev_track: 'previous',
};

async function mediaControlLinux(command: string): Promise<void> {
  await runExecFile('playerctl', [LINUX_COMMANDS[command]!]);
}

// macOS NX media key codes via System Events (requires Accessibility permission)
const MAC_KEY_CODES: Record<string, string> = {
  play_pause: '100',  // NX_KEYTYPE_PLAY (F8 media key)
  next_track: '101',  // NX_KEYTYPE_NEXT (F9 media key)
  prev_track: '98',   // NX_KEYTYPE_PREVIOUS (F7 media key)
};

async function mediaControlMac(command: string): Promise<void> {
  const keyCode = MAC_KEY_CODES[command]!;
  try {
    await runExecFile('osascript', [
      '-e',
      `tell application "System Events" to key code ${keyCode}`,
    ]);
  } catch (err) {
    const msg = err instanceof Error ? err.message : String(err);
    if (msg.includes('not allowed') || msg.includes('permission_denied') || msg.includes('access for assistive')) {
      throw new ActionValidationError(
        'permission_denied',
        'media_control requires Accessibility permission — grant JARVIS access in System Settings → Privacy & Security → Accessibility'
      );
    }

  prev_track: 177,  // VK_MEDIA_PREV_TRACK

  try {

    await runExecFile('powershell.exe', [
      '-NoProfile', '-Command',
      `(New-Object -ComObject WScript.Shell).SendKeys([char]${charCode})`,
    ]);

    // Fallback: nircmd sendkeypress
    const nircmdKey = command === 'play_pause' ? 'media_play_pause'
      : command === 'next_track' ? 'media_next_track'
      : 'media_prev_track';
    try {
      await runExecFile('nircmd.exe', ['sendkeypress', nircmdKey]);
    } catch {
      throw new Error('Unable to send media key — PowerShell WScript.Shell failed and nircmd not found. Install nircmd as fallback.');
    }
  }
```
