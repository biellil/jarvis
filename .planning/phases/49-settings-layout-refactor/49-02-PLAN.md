**Structure:**
```tsx
import { Slider, Button, Field } from '../../components/ui';
import type { SettingsSectionProps } from '../SettingsLayout';

type Props = Pick<SettingsSectionProps, 'vadThresholdMs' | 'onVadThresholdChange' | 'onVadThresholdReset'>;

export function AlwaysListeningSection({ vadThresholdMs, onVadThresholdChange, onVadThresholdReset }: Props) {
  return (
    <div>
      <h1 className="text-lg font-semibold text-fg tracking-tight">Voice Activity Detection</h1>
      <p className="text-sm text-fg-subtle mt-sm mb-lg">Tune voice activity detection sensitivity.</p>

      <Field>
        <Field.Label>Silence Threshold</Field.Label>
        <Field.Control>
          <Slider
            min={VAD_THRESHOLD_MIN_MS}
            max={VAD_THRESHOLD_MAX_MS}
            step={VAD_THRESHOLD_STEP_MS}
            value={[vadThresholdMs]}
            onValueChange={(vals) => void onVadThresholdChange(vals[0]!)}
            aria-label="VAD silence threshold in milliseconds"
            aria-valuemin={VAD_THRESHOLD_MIN_MS}
            aria-valuemax={VAD_THRESHOLD_MAX_MS}
            aria-valuenow={vadThresholdMs}
            aria-valuetext={`${vadThresholdMs} milliseconds`}
          />
        </Field.Control>
        <Field.Helper>
          Silence threshold after speech ends. Lower = more responsive but may
          trigger on breathing/clicks. Higher = patient but may miss end of
          phrase. Typical: 400–600ms.
        </Field.Helper>
      </Field>

      {/* Value display */}
      <p className="text-xs text-fg-subtle mt-sm">{vadThresholdMs} ms</p>

      {/* Reset button — D-12 */}
      <Button
        variant="ghost"
        size="sm"
        onClick={onVadThresholdReset}
        className="mt-base"
      >
        Reset to Default (500ms)
      </Button>
    </div>
  );
}
```

**Critical**: The Phase 48 Slider is a Radix `SliderPrimitive.Root` wrapper — it takes `value: number[]` and `onValueChange: (value: number[]) => void`. Wrap scalar `vadThresholdMs` as `[vadThresholdMs]` and unwrap in the callback with `vals[0]!`.

**ARIA preservation** (critical for existing Vitest tests):
- `aria-label="VAD silence threshold in milliseconds"` — EXACT string (test `getByLabelText(/VAD silence threshold in milliseconds/i)` depends on this)
- `aria-valuetext="{N} milliseconds"` — test for `{vadThresholdMs} ms` display label also exists

The display of `{vadThresholdMs} ms` must render in the DOM (existing test: `screen.getByText(/650 ms/i)`).

**Reset button copy**: EXACT string "Reset to Default (500ms)" (existing test: `getByText(/Reset to Default \(500ms\)/i)`).

**No local state** — this component is purely presentational; all state lives in SettingsLayout.
