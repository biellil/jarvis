```typescript
/**
 * AlwaysListeningStrategy — coordinator no main process (D-01).
 *
 * Responsabilidades:
 *   - Enviar IPC 'always-listening:start/stop' para o renderer iniciar/parar AlwaysListeningEngine
 *   - Registrar handler para 'always-listening:utterance' — recebe WAV do renderer e despacha para
 *     voiceHandler.handleAudio (pipeline STT→LLM→TTS existente)
 *   - Emitir 'voiceMode:degraded' via VoiceModeManager se start() falhar (D-09)
 *   - NÃO roda o loop de áudio, VAD, ring buffer ou classifier — tudo isso está no renderer (D-01)
 *
 * @see CONTEXT.md D-01: loop de captura vive no renderer; main apenas coordena lifecycle via IPC
 * @see CONTEXT.md D-04: IPC é utterance-level apenas — 1 event por fala detectada e aprovada
 */
import { ipcMain, BrowserWindow } from 'electron';
import type { VoiceCaptureStrategy } from '../index.js';
import { IPC_CHANNELS, type AlwaysListeningUtterancePayload, type VoiceModeDegradedEvent, type VoiceMode } from '../../shared/ipc-types.js';
import { handleAudio, type VoiceHandlerDeps } from '../voiceInput/voiceHandler.js';
import { getVadSilenceThresholdMs, setVadSilenceThresholdMs } from '../store.js';

export interface AlwaysListeningStrategyDeps {
  mainWindow: BrowserWindow;
  voiceHandlerDeps: VoiceHandlerDeps;
  onDegraded: (event: VoiceModeDegradedEvent) => void; // Callback para VoiceModeManager emitir evento
}

export class AlwaysListeningStrategy implements VoiceCaptureStrategy {
  private status: 'idle' | 'capturing' | 'processing' = 'idle';
  private utteranceHandler: ((event: Electron.IpcMainEvent, payload: AlwaysListeningUtterancePayload) => void) | null = null;
  private vadThresholdHandler: ((event: Electron.IpcMainInvokeEvent, ms: number) => Promise<{ success: boolean }>) | null = null;

  constructor(private readonly deps: AlwaysListeningStrategyDeps) {}

  async start(): Promise<void> {
    // Registrar handler para utterances do renderer (D-04: utterance-level IPC)
    this.utteranceHandler = (_, payload: AlwaysListeningUtterancePayload) => {
      void this.processUtterance(payload);
    };
    ipcMain.on(IPC_CHANNELS.ALWAYS_LISTENING_UTTERANCE, this.utteranceHandler);

    // Registrar handler para VAD threshold runtime apply (VLISTEN-04)
    this.vadThresholdHandler = async (_, ms: number) => {
      const clamped = Math.max(300, Math.min(800, ms)); // T-40-VAD: clamp defensivo
      setVadSilenceThresholdMs(clamped);
      // Broadcast de volta ao renderer para reconfigureVadThreshold() (Open Question A5)
      this.deps.mainWindow.webContents.send('vad:threshold-changed', clamped);
      return { success: true };
    };
    ipcMain.handle(IPC_CHANNELS.ALWAYS_LISTENING_VAD_THRESHOLD, this.vadThresholdHandler);

    // Enviar sinal para o renderer iniciar o AlwaysListeningEngine
    const vadThresholdMs = getVadSilenceThresholdMs();
    const negativeFramesToClose = Math.floor((vadThresholdMs / 1000) * 16000 / 1536); // 1536 = Silero frame size
    this.deps.mainWindow.webContents.send(IPC_CHANNELS.ALWAYS_LISTENING_START, { negativeFramesToClose });

    this.status = 'capturing';
  }

  private async processUtterance(payload: AlwaysListeningUtterancePayload): Promise<void> {
    if (this.status === 'idle') return; // já parado

    this.status = 'processing';

    try {
      await handleAudio(payload.wavBuffer, this.deps.voiceHandlerDeps);
    } catch (err) {
      console.error('[AlwaysListeningStrategy] handleAudio error:', err);
    } finally {
      // Retornar a 'capturing' se ainda ativo
      if (this.status === 'processing') {
        this.status = 'capturing';
      }
    }
  }

  async stop(): Promise<void> {
    this.status = 'idle';

    // Remover handlers IPC
    if (this.utteranceHandler) {
      ipcMain.off(IPC_CHANNELS.ALWAYS_LISTENING_UTTERANCE, this.utteranceHandler);
      this.utteranceHandler = null;
    }
    if (this.vadThresholdHandler) {
      ipcMain.removeHandler(IPC_CHANNELS.ALWAYS_LISTENING_VAD_THRESHOLD);
      this.vadThresholdHandler = null;
    }

    // Sinalizar renderer para parar o engine
    if (!this.deps.mainWindow.isDestroyed()) {
      this.deps.mainWindow.webContents.send(IPC_CHANNELS.ALWAYS_LISTENING_STOP);
    }
  }

  async dispose(): Promise<void> {
    await this.stop();
    // Strategy descartada — VoiceModeManager não re-usa esta instância
  }

  getStatus(): 'idle' | 'capturing' | 'processing' {
    return this.status;
  }
}
```

```typescript
/**
 * scheduleModelPreDownload — baixa o modelo multilingual-e5-small em background após app.whenReady().
 * Off no processo renderer (Transformers.js lazy download); aqui apenas sinaliza o renderer.
 * D-15: pre-download silencioso. D-16: falha emite voiceMode:degraded mas não bloqueia startup.
 */
export function scheduleModelPreDownload(
  mainWindow: BrowserWindow,
  onFail: (event: VoiceModeDegradedEvent) => void,
): void {
  // Dar tempo ao renderer para carregar antes de iniciar o download (5s após ready)
  setTimeout(() => {
    if (mainWindow.isDestroyed()) return;

    // Sinalizar renderer para iniciar download do modelo em background
    mainWindow.webContents.send('always-listening:preload-model');

    // Ouvir resposta de falha do renderer (se download falhar)
    ipcMain.once('always-listening:model-download-failed', () => {
      onFail({
        attemptedMode: 'always-listening' as VoiceMode,
        reason: 'classifier-download-fail',
        message: 'Sem conexão para baixar classifier de intent. Tente Always-Listening quando estiver online.',
      });
    });
  }, 5000);
}
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-40-VAD | Tampering | VAD threshold IPC handler | mitigate | `Math.max(300, Math.min(800, ms))` clamp no handler antes de setVadSilenceThresholdMs — mesmo que renderer envie valor fora de range, store recebe valor seguro |
| T-40-DEGRADE | Denial of Service | start() → classifier load fail | mitigate | Falha em start() propaga para VoiceModeManager que mantém modo anterior (`setMode()` retorna false); `onDegraded` emite IPC VOICE_MODE_DEGRADED para Phase 41 mostrar toast; sem crash silencioso |
| T-40-MODEL-DL | Tampering | scheduleModelPreDownload | mitigate | Timeout de 5s antes de iniciar DL (app já estável); falha emite event estruturado (não crash); usuário pode tentar novamente no próximo switch para always-listening (D-16) |

| T-40-RING | Information Disclosure | processUtterance dispose | accept | WAV buffer processado por handleAudio (STT) e descartado — não persiste em memória após processamento; sem log de áudio (D-12 off by default) |
