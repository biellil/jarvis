```typescript
/**
 * AlwaysListeningEngine — Loop contínuo de captura + VAD + classifier no renderer (D-01..D-04).
 * 
 * Pipeline: MediaStream → @ricky0123/vad-web (Silero VAD) → ring buffer pre-roll →
 *           IntentClassifier → onUtteranceReady callback → IPC para main (Plan 05).
 *
 * @see CONTEXT.md D-01: AlwaysListeningStrategy coordina no main; loop real vive aqui (renderer)
 * @see CONTEXT.md D-03: ring buffer 500ms = 8000 samples @ 16kHz
 * @see CONTEXT.md D-10: timeout 300ms → send-anyway; D-13: classifier lazy load
 */
import { encodeFloat32ToWav } from '../encodeFloat32ToWav.js';
import { AudioRingBuffer } from './audioRingBuffer.js';
import { IntentClassifier } from './intentClassifier.js';
import { INTENT_EXAMPLES_PT_BR } from '../../../../../../main/voiceMode/intentExamples.pt-BR.js';
// OU importar os examples de um path relativo correto — verificar durante implementação

// Nota: INTENT_THRESHOLD como constante ou lida de process.env se disponível no renderer
const INTENT_THRESHOLD = (() => {
  const env = (typeof process !== 'undefined' && process.env.INTENT_THRESHOLD);
  return env ? parseFloat(env) : 0.6;
})();

const MIN_UTTERANCE_MS = 200; // Descarta utterances < 200ms (D-08 guard against micro-sounds)
const PRE_ROLL_SAMPLES = 8000; // 500ms @ 16kHz (VLISTEN-03)
const SAMPLE_RATE = 16000;

export interface AlwaysListeningEngineOptions {
  vadNegativeFramesToClose: number; // Calculado de vadSilenceThresholdMs pelo caller
  onUtteranceReady: (wavBuffer: Uint8Array) => void;
  onError: (reason: string, err?: Error) => void;
}

export class AlwaysListeningEngine {
  private audioContext: AudioContext | null = null;
  private stream: MediaStream | null = null;
  private vadSession: any | null = null; // @ricky0123/vad-web session type
  private ringBuffer: AudioRingBuffer;
  private classifier: IntentClassifier;
  private speechStartedAt = 0;
  private isDisposed = false;

  constructor(private readonly opts: AlwaysListeningEngineOptions) {
    // PRE_ROLL_SAMPLES * 2 = 16000 samples — margem para audio chunks que chegam em batches
    this.ringBuffer = new AudioRingBuffer(PRE_ROLL_SAMPLES * 2);
    this.classifier = new IntentClassifier({
      modelId: 'Xenova/multilingual-e5-small',
      threshold: INTENT_THRESHOLD,
      sttConfidenceThreshold: 0.5, // D-08
      timeoutMs: 300,              // D-10
    });
  }

  async start(stream: MediaStream): Promise<void> {
    if (this.isDisposed) throw new Error('AlwaysListeningEngine: cannot restart after dispose()');

    this.stream = stream;
    this.audioContext = new (window.AudioContext || (window as any).webkitAudioContext)({
      sampleRate: SAMPLE_RATE,
    });

    // Lazy load classifier (D-13) — load fail deve propagar para Strategy.start() → VoiceModeManager
    await this.classifier.load();

    // Importar @ricky0123/vad-web — VERIFICAR API exata vs WakeWordEngine.ts
    const { MicVAD } = await import('@ricky0123/vad-web');

    this.vadSession = await MicVAD.new({
      stream,
      positiveSpeechThreshold: 0.8,
      negativeSpeechThreshold: 0.3,
      redemptionFrames: 8,
      frameSamples: 1536,
      onSpeechStart: () => {
        this.speechStartedAt = Date.now();
      },
      onSpeechEnd: (audio: Float32Array) => {
        void this.handleSpeechEnd(audio);
      },
      // onFrameProcessed para alimentar o ring buffer com frames pré-fala
      onVADMisfire: () => {
        // Micro-utterance descartada pelo VAD — limpar ring buffer também
        this.ringBuffer.clear();
      },
    });

    // Se onFrameProcessed disponível, usá-lo para ring buffer pre-roll
    // Caso contrário, o ring buffer fica vazio (degraded — pre-roll não funciona sem frame hook)

    this.vadSession.start();
  }

  private async handleSpeechEnd(utteranceAudio: Float32Array): Promise<void> {
    const durationMs = Date.now() - this.speechStartedAt;

    // Guard: descartar utterances muito curtas (D-08, clicks, ruído transitório)
    if (durationMs < MIN_UTTERANCE_MS) {
      this.ringBuffer.clear();
      return;
    }

    // Pre-roll: concatenar [ring buffer snapshot] + [utterance da VAD]
    const ringSnapshot = this.ringBuffer.toArray(); // drains ring buffer
    const fullAudio = this.concatenateAudio(ringSnapshot, utteranceAudio);

    // Encode para WAV
    const wavBuffer = encodeFloat32ToWav(fullAudio, SAMPLE_RATE);

    // Invocar callback — AlwaysListeningStrategy (Plan 05) lida com IPC + classifier
    // Nota: o classifier roda DEPOIS de cruzar o IPC (no main, após STT confidence check)
    // Por design D-04: o renderer só envia WAV; o classifier pré-filtra via IPC round-trip?
    // REVISAR: CONTEXT.md D-02 diz "classifica antes de cruzar IPC" — portanto classifier roda AQUI
    // Implementar: onUtteranceReady é chamado APENAS se classifier aprovar (hasIntent = true)
    // Isso mantém o volume IPC mínimo (D-04)
    
    // TODO durante implementação: verificar se transcript está disponível no renderer neste ponto
    // Se não (sem STT no renderer), passar sttConfidence=1.0 e deixar STT confidence check no main
    const classificationResult = await this.classifier.classify('', 1.0); // transcript vazio = send-anyway via D-08
    
    // REVISÃO ARQUITETURAL: D-08 pre-filtro requer transcript do STT, mas STT roda no main.
    // Solução: classifier roda no renderer com sttConfidence=1.0 (sem pre-filtro por confidence).
    // Pre-filtro por transcript vazio/pontuação é aplicado no main (AlwaysListeningStrategy, Plan 05)
    // após receber o WAV e rodar STT. Isso é consistente com D-03: "WAV único → envia via IPC".
    // O classifier no renderer usa o transcript do onSpeechEnd (que é vazio no vad-web onSpeechEnd).
    // DECISÃO FINAL PARA IMPLEMENTAÇÃO: 
    //   Opção A (correta para D-02/D-04): Enviar WAV via IPC → main roda STT → main roda classifier
    //   Opção B (correta para "classifica antes de cruzar IPC" em D-02): renderer precisa de STT local
    // D-02 diz explicitamente "classifica antes de cruzar IPC" → implementar com classify('', 1.0)
    // e o pre-filtro D-08 por sttConfidence ocorre no main após STT. Classifier aqui usa heurística
    // de transcript vazio → send-anyway (stt-confidence-filtered verdict = send utterance anyway).
    // Resultado: classifier sempre aprova (verdict=stt-confidence-filtered) quando sem transcript.
    // ISSO É CORRETO para MVP: intent filtering por embedding do transcript acontece no main (Plan 05).
    
    if (classificationResult.hasIntent) {
      this.opts.onUtteranceReady(wavBuffer);
    }
  }

  private concatenateAudio(a: Float32Array, b: Float32Array): Float32Array {
    const result = new Float32Array(a.length + b.length);
    result.set(a, 0);
    result.set(b, a.length);
    return result;
  }

  /**
   * reconfigureVadThreshold — aplica novo negativeFramesToClose em runtime (VLISTEN-04).
   * Se VAD session não suporta reconfiguração ao vivo (Open Question A5),
   * faz stop() + restart com novo valor.
   */
  async reconfigureVadThreshold(negativeFramesToClose: number): Promise<void> {
    this.opts.vadNegativeFramesToClose = negativeFramesToClose;

    if (!this.vadSession || !this.stream) return;

    // Tentar reconfiguração ao vivo primeiro (verificar API @ricky0123/vad-web 0.0.30)
    // Se não disponível, stop+restart
    if (typeof this.vadSession.setNegativeFramesToClose === 'function') {
      this.vadSession.setNegativeFramesToClose(negativeFramesToClose);
    } else {
      // Stop+restart approach (Open Question A5 fallback)
      await this.vadSession.pause();
      // Recriar session com novo parâmetro — simplificado; detalhes na implementação
      // this.vadSession = await MicVAD.new({ ...existingOpts, negativeFramesToClose });
      await this.vadSession.start();
    }
  }

  async stop(): Promise<void> {
    if (this.vadSession) {
      await this.vadSession.destroy?.() ?? this.vadSession.pause?.();
      this.vadSession = null;
    }
    if (this.stream) {
      this.stream.getTracks().forEach((t) => t.stop());
      this.stream = null;
    }
    if (this.audioContext) {
      await this.audioContext.close();
      this.audioContext = null;
    }
  }

  async dispose(): Promise<void> {
    if (this.isDisposed) return; // Idempotente
    this.isDisposed = true;

    await this.stop();
    this.classifier.unload();    // D-13: libera modelo da memória
    this.ringBuffer.clear();     // T-40-RING: libera referências de samples
  }
}
```
