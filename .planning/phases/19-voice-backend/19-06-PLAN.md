export interface VoiceHandlerOptions {
  session: ChatSession;
  stt: STTProvider;
  tts: TTSProvider;
  store: MemoryStore;
  conversationId?: number | null; // se null, audit sem FK
}

export interface VoiceHandlerResult {
  transcription: string;
  message: string;
  audio: Buffer;
  audioFormat: 'mp3' | 'wav' | 'opus';
  sttProvider: string;
  ttsProvider: string;
}

export class VoiceHandler {
  constructor(opts: VoiceHandlerOptions);
  async handle(audioBuffer: Buffer): Promise<VoiceHandlerResult>;
}
```

Python reference fluxo (chat.py linhas 85+):
1. Validar audio não-vazio → 400
2. Transcribe → se vazio → 400
3. session.send(transcription) dentro de lock (lock é externo aqui)
4. [novo] TTS synthesize(reply)
5. Retornar { transcription, message, audio_base64, ... }

Fluxo com audit:
1. t0 = Date.now()
2. Validar audioBuffer.length > 0 → throw Error('empty audio') com code 'EMPTY_AUDIO'
3. transcription = await stt.transcribe(audioBuffer)
4. sttLatencyMs = Date.now() - t0
5. Se transcription vazio/whitespace → log voice_calls success=0 error='empty transcription' → throw Error('no speech detected') code 'NO_SPEECH'
6. id = store.logVoiceCall({convId, audioBytes, transcription, sttProvider: stt.name, sttLatencyMs, success: true})
7. t1 = Date.now()
8. message = await session.send(transcription) — pode throw (propagar, audit já está gravado)
9. t2 = Date.now()
10. ttsResult = await tts.synthesize(message) — pode throw (FallbackTTSProvider já tentou ambos)
    - Se throw: updateVoiceCall(id, {success: 0, error}) → rethrow
11. ttsLatencyMs = Date.now() - t2
12. updateVoiceCall(id, {ttsProvider: ttsResult.providerUsed ?? tts.name, ttsLatencyMs, ttsBytes: ttsResult.audio.length, success: 1})
13. return { transcription, message, audio: ttsResult.audio, audioFormat: ttsResult.format, sttProvider: stt.name, ttsProvider: ttsResult.providerUsed ?? tts.name }
```
