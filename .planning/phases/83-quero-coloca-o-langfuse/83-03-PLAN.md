```typescript
import { Langfuse } from "@langfuse/core";

// Module-level singleton (reused across requests — this is correct for manual spans,
// unlike CallbackHandler which must be per-request):
const langfuse = config.langfuseEnabled
  ? new Langfuse({
      publicKey: config.langfusePublicKey,
      secretKey: config.langfuseSecretKey,
      baseUrl: config.langfuseHost,
    })
  : null;

// Usage:
const span = langfuse?.span({
  name: "operation:name",
  input: { key: value },
});
try {
  // ... operation ...
  span?.end({ output: { result: summary } });
} catch (err) {
  span?.end({ status: "error", error: (err as Error).message });
  throw err; // or return fallback, depending on existing error handling
}
```
