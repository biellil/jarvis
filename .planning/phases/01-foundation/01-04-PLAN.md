```python
import sys
import importlib.metadata
import httpx
from packaging.version import Version

VERSION_PINS = {
    "langchain-core": "1.2.22",
    "langgraph-checkpoint-sqlite": "3.0.1",
}


def validate_versions() -> None:
    for package, minimum in VERSION_PINS.items():
        try:
            installed = importlib.metadata.version(package)
            if Version(installed) < Version(minimum):
                print(
                    f"ERRO: {package} {installed} instalado, mas >={minimum} e necessario. "
                    f"Execute: pip install '{package}>={minimum}'"
                )
                sys.exit(1)
        except importlib.metadata.PackageNotFoundError:
            print(
                f"ERRO: Pacote '{package}' nao encontrado. "
                f"Execute: pip install '{package}>={minimum}'"
            )
            sys.exit(1)


def validate_lm_studio_reachable(url: str) -> None:
    try:
        with httpx.Client(timeout=3.0) as client:
            client.get(f"{url}/models")
    except (httpx.ConnectError, httpx.TimeoutException):
        print(
            f"ERRO: LM Studio nao acessivel em {url}. "
            "Verifique se esta rodando e se LM_STUDIO_URL esta correto."
        )
        sys.exit(1)
```

**src/jarvis/core/session.py** per RESEARCH.md Pattern 3 and D-02:
```python
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage, BaseMessage
from langchain_core.language_models.chat_models import BaseChatModel

SYSTEM_PROMPT = "You are JARVIS, a helpful personal assistant."


class ChatSession:
    def __init__(self, llm: BaseChatModel):
        self.llm = llm
        self.history: list[BaseMessage] = [SystemMessage(content=SYSTEM_PROMPT)]

    async def send(self, user_input: str) -> str:
        self.history.append(HumanMessage(content=user_input))
        full_response = ""
        async for chunk in self.llm.astream(self.history):
            token = chunk.content
            if token:
                print(token, end="", flush=True)  # per D-02: plain print, no Rich
                full_response += token
        print()  # newline after stream
        self.history.append(AIMessage(content=full_response))
        return full_response
```

Per D-02: Token streaming uses `print(token, end='', flush=True)` — NO Rich on output.

```python
import asyncio
import sys
from rich.console import Console
from rich.prompt import Prompt
from jarvis.config import settings
from jarvis.llm.factory import create_llm
from jarvis.llm.capabilities import detect_capabilities
from jarvis.core.startup import validate_versions, validate_lm_studio_reachable
from jarvis.core.session import ChatSession

console = Console()


def show_banner(caps=None):
    """Display startup banner per D-14: ASCII + status block."""
    console.print("\n[bold cyan]╔══════════════════════════════════════╗[/bold cyan]")
    console.print("[bold cyan]║         J.A.R.V.I.S.  v0.1.0        ║[/bold cyan]")
    console.print("[bold cyan]╚══════════════════════════════════════╝[/bold cyan]\n")

    console.print(f"  [dim]Provider:[/dim]  [bold]{settings.llm_provider}[/bold]")
    model_name = settings.lm_studio_model or settings.llm_model or "(default)"
    console.print(f"  [dim]Model:[/dim]     [bold]{model_name}[/bold]")

    if caps:
        console.print(f"  [dim]Tools:[/dim]     {'[green]yes[/green]' if caps.tool_calling else '[red]no[/red]'}")
        console.print(f"  [dim]Vision:[/dim]    {'[green]yes[/green]' if caps.vision else '[red]no[/red]'}")
        ctx_str = str(caps.context_window) if caps.context_window else "unknown"
        console.print(f"  [dim]Context:[/dim]   [bold]{ctx_str}[/bold]")

    console.print()


async def main_async() -> None:
    # ARCH-04: Validate dependency versions
    validate_versions()

    # ARCH-04: Validate LM Studio reachability (only for lmstudio provider)
    if settings.llm_provider == "lmstudio":
        validate_lm_studio_reachable(settings.lm_studio_url)

    # LLM-01: Create LLM via factory
    llm = create_llm()

    # LLM-02: Detect capabilities for banner
    caps = None
    model_id = settings.lm_studio_model or settings.llm_model
    if model_id:
        caps = detect_capabilities(settings.lm_studio_url, model_id)

    # D-14: Show banner
    show_banner(caps)

    # CONV-01: Create session and enter conversation loop
    session = ChatSession(llm)
    console.print("[dim]Digite 'exit' ou 'quit' para sair. Ctrl+C tambem funciona.[/dim]\n")

    while True:
        try:
            # D-01: Rich-styled prompt
            user_input = Prompt.ask("[bold green]Voce[/bold green]")
        except (KeyboardInterrupt, EOFError):
            # D-04: Ctrl+C exits gracefully
            console.print("\n[dim]Encerrando...[/dim]")
            break

        if not user_input.strip():
            continue

        # D-04: exit/quit exits
        if user_input.strip().lower() in ("exit", "quit"):
            console.print("[dim]Ate logo![/dim]")
            break

        # D-03: Rich label for JARVIS, then plain streaming (D-02)
        console.print("[bold cyan]JARVIS:[/bold cyan] ", end="")
        await session.send(user_input)


def main() -> None:
    try:
        asyncio.run(main_async())
    except KeyboardInterrupt:
        pass


if __name__ == "__main__":
    main()
```
