    3. **src/jarvis/executor/base.py** — ActionExecutor class:

    ```python
    """ActionExecutor — interprets tool payloads and dispatches to Linux handlers.
    
    Per D-02: @tool functions return payloads; ActionExecutor executes them.
    Per D-03: Destructive actions require user confirmation via asyncio.to_thread(input).
    Per TOOL-05: Every execution is logged via ToolLogger.
    """
    import asyncio
    from typing import Optional

    from loguru import logger

    from jarvis.executor.linux import (
        handle_list_files, handle_search_files, handle_move_file, handle_delete_file,
        handle_open_app, handle_close_app,
        handle_set_volume, handle_set_brightness, handle_list_processes,
    )
    from jarvis.memory.store import ToolLogger

    class ActionExecutor:
        def __init__(self, tool_logger: ToolLogger, confirm_callback=None) -> None:
            self._logger = tool_logger
            self._confirm = confirm_callback  # async callable(tool_name, params) -> bool
            self._handlers = {
                "list_files": handle_list_files,
                "search_files": handle_search_files,
                "move_file": handle_move_file,
                "delete_file": handle_delete_file,
                "open_app": handle_open_app,
                "close_app": handle_close_app,
                "set_volume": handle_set_volume,
                "set_brightness": handle_set_brightness,
                "list_processes": handle_list_processes,
            }

        async def execute(self, tool_name: str, payload: dict, params: dict) -> dict:
            # D-03: confirmation for destructive actions
            if payload.get("requires_confirmation"):
                if self._confirm:
                    confirmed = await self._confirm(tool_name, params)
                else:
                    confirmed = await self._default_confirm(tool_name, params)
                if not confirmed:
                    self._logger.log(tool_name, params, "cancelled")
                    return {"status": "cancelled", "message": "Acao cancelada pelo usuario."}

            handler = self._handlers.get(payload.get("action"))
            if handler is None:
                err = f"Acao desconhecida: {payload.get('action')}"
                self._logger.log(tool_name, params, "error", err)
                return {"status": "error", "message": err}

            try:
                result = handler(payload.get("args", {}))
                self._logger.log(tool_name, params, "success")
                return result
            except Exception as e:
                logger.warning(f"ActionExecutor error ({tool_name}): {e}")
                self._logger.log(tool_name, params, "error", str(e))
                return {"status": "error", "message": str(e)}

        async def _default_confirm(self, tool_name: str, params: dict) -> bool:
            desc = f"{tool_name}: {params}"
            print(f"\n[confirmacao]: Vou executar {desc}. Confirma? (s/n)")
            response = await asyncio.to_thread(input, "")
            return response.strip().lower() in ("s", "sim", "y", "yes")
    ```

    Note: `confirm_callback` is injectable for testing (pass an async function that returns True/False). Default uses `asyncio.to_thread(input)` per ARCH-02.
