    const SENTENCE_BOUNDARY = /[.!?]\s+/;

    export class SentenceChunker {
      private buffer = '';
      feed(text: string): string[] {
        this.buffer += text;
        const out: string[] = [];
        let match: RegExpExecArray | null;
        const re = /[.!?]\s+/g;
        // Use exec in a loop on a fresh regex each iteration to avoid lastIndex pitfalls.
        while ((match = /[.!?]\s+/.exec(this.buffer)) !== null) {
          const end = match.index + match[0].length;
          const sentence = this.buffer.slice(0, end).trim();
          if (sentence.length > 0) out.push(sentence);
          this.buffer = this.buffer.slice(end);
        }
        return out;
      }
      flush(): string[] {
        const trimmed = this.buffer.trim();
        this.buffer = '';
        return trimmed.length > 0 ? [trimmed] : [];
      }
    }
    ```

    import { randomUUID } from 'node:crypto';
    import { BrowserWindow } from 'electron';
    import { SentenceChunker } from './chunker';
    import { openChatStream } from '../sse-client';
    import { getActiveTtsProvider } from './tts';
    import { IPC_CHANNELS, type TTSChunkPayload } from '../../shared/ipc-types';

    export interface StreamingTurnDeps {
      backendUrl: string;
      apiKey?: string;
      mainWindow: BrowserWindow;
      // Optional override for testing
      openStream?: typeof openChatStream;
      provider?: ReturnType<typeof getActiveTtsProvider>;
    }

    export interface StreamingTurnHandle {
      turnId: string;
      abort: () => void;
      done: Promise<void>;
    }

    export function runStreamingTurn(
      deps: StreamingTurnDeps,
      transcription: string,
    ): StreamingTurnHandle {
      const turnId = randomUUID();
      const controller = new AbortController();
      const cancelled = { value: false };
      const chunker = new SentenceChunker();
      const tts = deps.provider ?? getActiveTtsProvider();
      const openStream = deps.openStream ?? openChatStream;
      let nextIdx = 0;
      const inFlight: Promise<void>[] = [];

      const sendChunk = (payload: TTSChunkPayload) => {
        if (cancelled.value) return;
        if (deps.mainWindow.isDestroyed()) return;
        deps.mainWindow.webContents.send(IPC_CHANNELS.TTS_CHUNK, payload);
      };

      const synthAndSend = async (text: string, idx: number): Promise<void> => {
        if (cancelled.value) return;
        try {
          const result = await tts.synthesize(text);
          if (cancelled.value) return;
          sendChunk({
            turnId,
            idx,
            audioBase64: result.audio.toString('base64'),
            format: result.format,
            isLast: false,
          });
        } catch (err) {
          // Graceful degrade per WAKE-10 precedent — skip failed sentence
          console.warn(`[streaming-tts] sentence ${idx} synthesize failed:`, err);
        }
      };

      const done = (async () => {
        try {
          await openStream({
            url: `${deps.backendUrl}/api/chat/stream`,
            apiKey: deps.apiKey,
            message: transcription,
            onToken: (tok: string) => {
              for (const sentence of chunker.feed(tok)) {
                inFlight.push(synthAndSend(sentence, nextIdx++));
              }
            },
            onAction: () => { /* Phase 54 territory */ },
            onEnd: () => { /* drain handled below */ },
            onError: (err) => console.warn('[streaming-tts] sse error:', err),
            signal: controller.signal,
          });
        } catch (err) {
          if (!cancelled.value) console.warn('[streaming-tts] stream error:', err);
        }
        // Flush residual sentence (D-04)
        for (const tail of chunker.flush()) {
          inFlight.push(synthAndSend(tail, nextIdx++));
        }
        await Promise.allSettled(inFlight);
        if (cancelled.value) return;
        if (deps.mainWindow.isDestroyed()) return;
        deps.mainWindow.webContents.send(IPC_CHANNELS.TTS_END, { turnId });
      })();

      return {
        turnId,
        abort: () => {
          cancelled.value = true;
          try { controller.abort(); } catch { /* already aborted */ }
        },
        done,
      };
    }
    ```

    - Per D-08, single code path for both Murf and ElevenLabs (just calls `tts.synthesize`).
