    export function createTTSProvider(): TTSProvider {
      // Phase 34: store values take precedence over env vars (SET-03)
      const storedProvider = getTtsProvider();  // 'murf' | 'elevenlabs'
      const storedApiKey = getTtsApiKey();      // '' if not set

      // Determine provider: store-set value (non-default) wins, then env var
      const raw = process.env["TTS_PROVIDER"];
      const provider = storedProvider !== 'elevenlabs'  // non-default store value takes priority
        ? storedProvider
        : (raw && raw.trim().length > 0 ? raw : "elevenlabs").toLowerCase();

      if (provider === "murf") {
        const apiKey = storedApiKey || process.env["MURF_API_KEY"];
        if (!apiKey) {
          console.warn("[voice] TTS_PROVIDER=murf but MURF_API_KEY not set, falling back to elevenlabs");
          return new ElevenLabsTTSProvider();
        }
        return new MurfTTSProvider(apiKey);  // pass key directly if MurfTTSProvider accepts it
      }

      const apiKey = storedApiKey || process.env["ELEVENLABS_API_KEY"];
      if (!apiKey) {
        console.warn("[voice] TTS_PROVIDER=elevenlabs but ELEVENLABS_API_KEY not set, falling back to murf");
        return new MurfTTSProvider();
      }
      return new ElevenLabsTTSProvider(apiKey);  // pass key directly if ElevenLabsTTSProvider accepts it
    }
