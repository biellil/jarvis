TypeScript targets:
```typescript
// profile.ts
export const EXPLICIT_TRIGGERS: readonly string[];
export function isExplicitProfileCommand(text: string): boolean;
export async function extractProfileFacts(
  llm: BaseChatModel,  // from @langchain/core/language_models/chat_models
  userInput: string,
): Promise<Record<string, string>>;

// manager.ts
export interface MemoryManagerOptions {
  dbPath?: string;          // defaults to ./data/jarvis.db
  chromaPath?: string;      // defaults to ./data/chroma
  recallTopK?: number;      // default 5
  recallThreshold?: number; // default 0.5
}

export class MemoryManager {
  constructor(opts?: MemoryManagerOptions);
  async startConversation(): Promise<number | null>;
  async endConversation(convId: number): Promise<void>;
  async saveTurn(convId: number, userText: string, assistantText: string): Promise<void>;
  async buildContext(userText: string): Promise<string>; // system-prompt-ready block
  async learnFromTurn(llm: BaseChatModel, userText: string): Promise<void>;
  getProfileFacts(): Array<{ key: string; value: string; source: string; createdAt: string }>;
  close(): void;
}
```

`buildContext` output format (string, injected as SystemMessage by ChatSession):
```
### User profile
- nome: Alice
- linguagem_trabalho: Python e Go

### Recall from past conversations
- "I prefer dark mode in everything"
- "My cat's name is Mochi"
```
Sections omitted when empty. Returns empty string if both empty.

LangChain.js import: `import type { BaseChatModel } from '@langchain/core/language_models/chat_models';` (already a transitive dep from phase 15).
