```typescript
import { Router, type Request, type Response } from 'express';
import { Command } from '@langchain/langgraph';
import { taskCheckpointer } from '../agent/graph.js';
import { resumeRequestSchema } from '../agent/types.js';
import { authMiddleware } from '../middleware/auth.js'; // path may differ; check apps/backend-ts/src/middleware/

// Module-level singletons — shared with routes/chat.ts via export
export const activeControllers = new Map<string, AbortController>();
export const activeGraphs = new Map<string, ReturnType<typeof import('../agent/graph.js').buildTaskGraph>>();

const TASK_ID_PATTERN = /^chat-[\w-]+-task-[0-9a-f-]{36}$/;

export function createTasksRouter(): Router {
  const router = Router();

  router.post('/:taskId/resume', authMiddleware, async (req: Request, res: Response) => {
    const { taskId } = req.params;
    if (!TASK_ID_PATTERN.test(taskId)) {
      return res.status(400).json({ error: 'taskId malformado' });
    }
    const graph = activeGraphs.get(taskId);
    if (!graph) {
      return res.status(404).json({ error: 'Task não encontrada ou já encerrada' });
    }

    const parsed = resumeRequestSchema.safeParse(req.body);
    if (!parsed.success) {
      return res.status(400).json({ error: 'Body inválido', issues: parsed.error.format() });
    }

    const controller = activeControllers.get(taskId) ?? new AbortController();
    activeControllers.set(taskId, controller);

    res.setHeader('Content-Type', 'text/event-stream');
    res.setHeader('Cache-Control', 'no-cache');
    res.setHeader('Connection', 'keep-alive');
    res.flushHeaders();

    try {
      const stream = await graph.stream(
        new Command({ resume: parsed.data }),
        {
          configurable: { thread_id: taskId },
          streamMode: ['custom', 'messages'] as const,
          signal: controller.signal,
        },
      );
      let isTerminal = false;
      for await (const [mode, chunk] of stream) {
        if (mode === 'custom') {
          const evt = chunk as { kind: string };
          res.write(`event: ${evt.kind}\ndata: ${JSON.stringify({ taskId, ...evt })}\n\n`);
          if (evt.kind === 'task:done' || evt.kind === 'task:cancelled' || evt.kind === 'task:error') {
            isTerminal = true;
          }
        } else if (mode === 'messages') {
          // Forward AIMessageChunk content as plain SSE data lines (matches existing chat path)
          const [msgChunk] = chunk as [{ content?: unknown }];
          const content = msgChunk?.content;
          if (typeof content === 'string' && content) {
            res.write(`data: ${content.replace(/\n/g, '\\n')}\n\n`);
          }
        }
      }
      // After stream drains, check if state is paused on a NEW interrupt (e.g., step-failure)
      const snapshot = await graph.getState({ configurable: { thread_id: taskId } });
      const interrupts = snapshot.tasks?.[0]?.interrupts ?? [];
      if (interrupts.length > 0) {
        // Step-failure interrupt — emit awaiting-failure-decision
        const interruptValue = interrupts[0].value as { kind: string; stepId?: number; error?: string };
        if (interruptValue.kind === 'step-failure') {
          res.write(
            `event: task:awaiting-failure-decision\ndata: ${JSON.stringify({
              taskId,
              stepId: interruptValue.stepId,
              error: interruptValue.error,
            })}\n\n`,
          );
        }
      } else if (isTerminal) {
        await taskCheckpointer.deleteThread(taskId);
        activeControllers.delete(taskId);
        activeGraphs.delete(taskId);
      }
    } catch (err) {
      const message = (err as Error).message ?? 'Erro desconhecido';
      res.write(`event: task:error\ndata: ${JSON.stringify({ taskId, atStep: 0, message })}\n\n`);
      await taskCheckpointer.deleteThread(taskId).catch(() => {});
      activeControllers.delete(taskId);
      activeGraphs.delete(taskId);
    } finally {
      res.end();
    }
  });

  router.post('/:taskId/cancel', authMiddleware, async (req: Request, res: Response) => {
    const { taskId } = req.params;
    if (!TASK_ID_PATTERN.test(taskId)) {
      return res.status(400).json({ error: 'taskId malformado' });
    }
    const controller = activeControllers.get(taskId);
    const graph = activeGraphs.get(taskId);
    if (!controller || !graph) {
      return res.status(404).json({ error: 'Task não encontrada' });
    }
    // D-13 — flip state.cancelRequested for cancel-gate AND abort active fetch operations
    await graph.updateState(
      { configurable: { thread_id: taskId } },
      { cancelRequested: true },
    );
    controller.abort();
    res.json({ status: 'cancelling' });
  });

  return router;
}
```
