    ```typescript
    /**
     * ProactiveScheduler — Phase 67 (D-02, D-03)
     *
     * Manages all proactive timing: per-reminder cron jobs + recurring daily summary.
     * Runs in backend-ts (always-live service).
     *
     * IMPORTANT: MemorySaver-style cleanup — jobs Map must be explicitly managed.
     * On terminal events (fired/cancelled), destroy() the job to prevent ghost fires.
     */
    import * as cron from 'node-cron';
    import { EventEmitter } from 'events';
    import { db } from '../memory/db.js';
    import { reminders } from '../memory/schema.js';
    import { eq, inArray } from 'drizzle-orm';
    import { isInQuietHours, nextQuietEnd } from './quiet-hours.js';
    import type { ProactiveEvent } from './types.js';

    /** Shared EventEmitter — routes/proactive.ts subscribes to 'event' to forward via SSE */
    export const proactiveEmitter = new EventEmitter();

    export class ProactiveScheduler {
      /** reminderId → scheduled cron task. Special key -1 = daily summary. */
      private static jobs = new Map<number, cron.ScheduledTask>();

      private static quietHours = {
        enabled: false,
        start: '22:00',
        end: '08:00',
      };

      /** Call after DB migrations in index.ts. Restores all pending/deferred jobs. */
      static bootstrap(): void {
        // SELECT all pending or deferred reminders
        const rows = db
          .select()
          .from(reminders)
          .where(inArray(reminders.status, ['pending', 'deferred']))
          .all();

        for (const row of rows) {
          const dueAt = row.deferred_until ?? row.due_at;
          ProactiveScheduler.registerJob(row.id, dueAt);
        }
      }

      /** Register one cron job for a specific reminder at dueAtMs epoch. */
      static registerJob(reminderId: number, dueAtMs: number): void {
        // Destroy any existing job for this id (re-schedule on deferred)
        ProactiveScheduler.jobs.get(reminderId)?.destroy();

        const due = new Date(dueAtMs);
        const minute = due.getMinutes();
        const hour = due.getHours();
        const day = due.getDate();
        const month = due.getMonth() + 1;
        // One-off: fires on the specific day+month, any year's matching weekday ignored
        const cronExpr = `${minute} ${hour} ${day} ${month} *`;

        const task = cron.schedule(
          cronExpr,
          () => { void ProactiveScheduler.fireReminder(reminderId); },
          { scheduled: true },
        );

        ProactiveScheduler.jobs.set(reminderId, task);
      }

      /** Fire a reminder: quiet-aware. Defers if in quiet, emits SSE event if not. */
      static async fireReminder(reminderId: number): Promise<void> {
        const row = db
          .select()
          .from(reminders)
          .where(eq(reminders.id, reminderId))
          .get();

        if (!row || row.status !== 'pending') {
          // Already fired, cancelled, or deferred by another path — idempotent
          return;
        }

        const now = new Date();

        if (
          ProactiveScheduler.quietHours.enabled &&
          isInQuietHours(now, ProactiveScheduler.quietHours.start, ProactiveScheduler.quietHours.end)
        ) {
          const deferredUntil = nextQuietEnd(now, ProactiveScheduler.quietHours.end);
          db.update(reminders)
            .set({ status: 'deferred', deferred_until: deferredUntil.getTime() })
            .where(eq(reminders.id, reminderId))
            .run();
          // Re-schedule for the deferred time
          ProactiveScheduler.registerJob(reminderId, deferredUntil.getTime());
        } else {
          db.update(reminders)
            .set({ status: 'fired', fired_at: Date.now() })
            .where(eq(reminders.id, reminderId))
            .run();
          // Emit SSE event (routes/proactive.ts subscriber)
          const event: ProactiveEvent = {
            kind: 'reminder',
            id: row.id,
            message: row.message,
            dueAt: row.due_at,
          };
          proactiveEmitter.emit('event', event);
          // Clean up job after firing
          ProactiveScheduler.jobs.get(reminderId)?.destroy();
          ProactiveScheduler.jobs.delete(reminderId);
        }
      }

      /** Cancel a reminder: stop its job + update DB status. */
      static cancelReminder(reminderId: number): void {
        ProactiveScheduler.jobs.get(reminderId)?.destroy();
        ProactiveScheduler.jobs.delete(reminderId);
        db.update(reminders)
          .set({ status: 'cancelled' })
          .where(eq(reminders.id, reminderId))
          .run();
      }

      /** Register or update the recurring daily summary job. */
      static registerDailySummaryJob(timeHHMM: string): void {
        ProactiveScheduler.jobs.get(-1)?.destroy();
        const [hStr, mStr] = timeHHMM.split(':');
        const h = parseInt(hStr ?? '9', 10);
        const m = parseInt(mStr ?? '0', 10);
        const cronExpr = `${m} ${h} * * *`;

        const task = cron.schedule(
          cronExpr,
          () => { void ProactiveScheduler.generateAndEmitDailySummary(); },
          { scheduled: true },
        );
        ProactiveScheduler.jobs.set(-1, task);
      }

      /**
       * Called by daily summary cron. Emits placeholder; actual generation is in
       * DailySummaryGenerator (Plan 67-06) which hooks into this via proactiveEmitter.
       * This method is overridden by Plan 67-06 to perform the LLM call.
       */
      static async generateAndEmitDailySummary(): Promise<void> {
        // Placeholder — overridden in Plan 67-06 by importing DailySummaryGenerator
        // and replacing this function's body. Keeping stub here keeps the scheduler
        // self-contained for testing.
        console.log('[ProactiveScheduler] Daily summary trigger (stub — Plan 67-06 implements body)');
      }

      /** Update quiet hours config at runtime (from Settings IPC). */
      static updateQuietHours(enabled: boolean, start: string, end: string): void {
        ProactiveScheduler.quietHours = { enabled, start, end };
      }

      /** Exposed for testing: get current job count. */
      static get jobCount(): number {
        return ProactiveScheduler.jobs.size;
      }
    }
    ```
