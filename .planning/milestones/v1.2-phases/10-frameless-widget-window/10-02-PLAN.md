**Option B: Use Node.js script with canvas package** (if ImageMagick not available):
```javascript
// scripts/generate-tray-icons.js
const { createCanvas } = require('canvas');
const fs = require('fs');

function createIcon(size, outputPath) {
  const canvas = createCanvas(size, size);
  const ctx = canvas.getContext('2d');

  const center = size / 2;
  const radius = size * 0.375; // 6px for 16x16, 12px for 32x32

  // Create radial gradient
  const gradient = ctx.createRadialGradient(
    center * 0.9, center * 0.9, 0,  // Inner circle (slightly offset for depth)
    center, center, radius
  );
  gradient.addColorStop(0, 'rgba(255, 255, 255, 0.6)');
  gradient.addColorStop(1, '#06B6D4');

  // Draw circle with gradient
  ctx.beginPath();
  ctx.arc(center, center, radius, 0, Math.PI * 2);
  ctx.fillStyle = gradient;
  ctx.fill();

  // Add subtle glow (outer shadow simulation)
  ctx.shadowColor = 'rgba(6, 182, 212, 0.4)';
  ctx.shadowBlur = size * 0.25;
  ctx.beginPath();
  ctx.arc(center, center, radius * 0.9, 0, Math.PI * 2);
  ctx.fillStyle = 'transparent';
  ctx.fill();

  // Save to PNG
  const buffer = canvas.toBuffer('image/png');
  fs.writeFileSync(outputPath, buffer);
  console.log(`Created: ${outputPath}`);
}

createIcon(16, 'apps/desktop/resources/tray/icon-16x16.png');
createIcon(32, 'apps/desktop/resources/tray/icon-32x32.png');
```

Create `apps/desktop/src/main/tray.ts`:

```typescript
/**
 * System Tray Module
 *
 * DESK-04: Tray icon with menu Show/Hide/Quit
 * D-05: Icon 16x16 + 32x32 PNG cyan circle
 * D-06: Single-click shows menu (Windows/Linux default)
 * D-07: Tooltip "JARVIS"
 * D-08: Menu with exactly 3 items: Show, Hide, Quit
 */
import { Tray, Menu, app, BrowserWindow } from 'electron';
import path from 'node:path';

let tray: Tray | null = null;

export function createTray(mainWindow: BrowserWindow): void {
  // D-05: Use 16x16 icon (Electron auto-selects 32x32 for high-DPI)
  const iconPath = path.join(__dirname, '../../resources/tray/icon-16x16.png');
  tray = new Tray(iconPath);

  // D-07: Simple tooltip with app name only
  tray.setToolTip('JARVIS');

  // D-08: Exactly 3 menu items for DESK-04 compliance
  const contextMenu = Menu.buildFromTemplate([
    {
      label: 'Show',
      click: () => {
        mainWindow.show();
      },
    },
    {
      label: 'Hide',
      click: () => {
        mainWindow.hide();
      },
    },
    {
      label: 'Quit',
      click: () => {
        app.quit();
      },
    },
  ]);

  // D-06: Single-click shows context menu (default behavior)
  tray.setContextMenu(contextMenu);
}

export function destroyTray(): void {
  if (tray) {
    tray.destroy();
    tray = null;
  }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Tray click -> App action | User clicking tray menu triggers window operations |
| Icon file -> Tray | PNG file loaded from disk at startup |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-04 | Tampering | Icon path | mitigate | Use path.join(__dirname, relative) with static filename; never accept user input for icon paths |
| T-10-05 | Spoofing | Tray menu | accept | Tray menu items are hardcoded; no dynamic menu generation from external input |
| T-10-06 | Elevation of Privilege | menu click handlers | accept | Menu handlers only call safe Electron APIs (show, hide, quit); no arbitrary code execution |
</threat_model>
