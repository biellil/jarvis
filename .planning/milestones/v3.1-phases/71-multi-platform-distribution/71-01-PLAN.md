    Create `apps/desktop/src/main/firstRunEnv.ts` with EXACTLY:

    ```typescript
    /**
     * firstRunEnv.ts — Idempotent first-run copy of .env.example → userData/.env.
     *
     * Phase 71 (D-08): runs BEFORE runLlmConfigMigration and BEFORE process.loadEnvFile.
     * Without this, packaged app has no .env to read/migrate into.
     *
     * Security (T-71-01): on POSIX, sets 0o600 permission so API keys are owner-only.
     * Windows: skipped (NTFS ACLs from %APPDATA%\JARVIS\ are already user-scoped).
     */
    import { app } from 'electron';
    import fs from 'node:fs';
    import path from 'node:path';
    import process from 'node:process';
    import { resolveEnvPath } from './envPath.js';

    /**
     * Ensures userData/.env exists in packaged builds.
     * Returns true if a copy occurred, false otherwise (dev mode, file already present, template missing).
     */
    export function ensureUserEnvFile(): boolean {
      // Dev: no-op (workflow already places .env at monorepo root)
      if (!app.isPackaged) return false;

      const envPath = resolveEnvPath();

      // Idempotent guard — never overwrite existing user .env
      if (fs.existsSync(envPath)) return false;

      const template = path.join(process.resourcesPath, '.env.example');
      if (!fs.existsSync(template)) {
        console.warn(`[first-run] template not found at ${template} — skipping .env copy`);
        return false;
      }

      try {
        fs.mkdirSync(path.dirname(envPath), { recursive: true });
        fs.copyFileSync(template, envPath);
        console.log(`[first-run] .env created at ${envPath} from template`);

        // T-71-01: POSIX-only chmod 0o600 (owner read/write, others denied)
        if (process.platform !== 'win32') {
          fs.chmodSync(envPath, 0o600);
          console.log(`[first-run] .env mode set to 0o600 (owner-only)`);
        }
        return true;
      } catch (err) {
        console.error(`[first-run] failed to copy .env from template:`, err);
        return false;
      }
    }
    ```

<threat_model>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| OS filesystem ↔ Electron main | First-run copy writes to `userData/.env` (Electron-controlled path, no user input on path) |
| Bundled resources ↔ runtime | `.env.example` read from `process.resourcesPath` (read-only, OS-protected install dir) |
| Tray click ↔ shell.showItemInFolder | `shell.showItemInFolder(path)` accepts a string from `resolveEnvPath()` — no user-controlled input |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-71-01 | Information Disclosure | `userData/.env` containing API keys readable by other OS users | mitigate | `firstRunEnv.ts` calls `fs.chmodSync(envPath, 0o600)` after copy on POSIX (Linux/macOS). Windows: NTFS ACL inheritance from `%APPDATA%\JARVIS\` is already user-scoped (skipped). Test asserts chmod is called on POSIX and skipped on win32. |
| T-71-01a | Information Disclosure | `.env` exists pre-Phase 71 (P70 migration ran with looser umask) | accept | Existing files retain pre-Phase 71 permissions. Acceptable: dev-only flow, and P71 first-run only triggers when file missing. Documentation in README notes this. |
| T-71-06 | Tampering | Path traversal via crafted `.env.example` symlink in bundled resources | mitigate | `process.resourcesPath` is in the OS-protected install dir, not user-writable. `fs.copyFileSync` resolves the source path at OS level — no path concatenation with user input. |
| T-71-07 | Tampering | Tray "Abrir .env" used to open arbitrary file | mitigate | `shell.showItemInFolder` receives path from `resolveEnvPath()` (computed function, no string concatenation with user input). Click handler has no parameters. |

</threat_model>
