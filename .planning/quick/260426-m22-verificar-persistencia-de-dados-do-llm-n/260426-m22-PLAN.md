**1) Criar `apps/backend-ts/src/routes/debug.ts`:**

Como o endpoint precisa do `db` do drizzle, adicionar parâmetro adicional para receber a instância (ou importar o módulo `db` direto se for um singleton no projeto — confirmar olhando como `health.ts` ou `tool-calls.ts` acessam o store).

Estrutura sugerida:
```typescript
import type { FastifyInstance } from 'fastify';
import { sql, desc } from 'drizzle-orm';
import { db } from '../memory/db'; // ajustar path conforme o singleton existente
import { conversations, messages, typedMemories, toolCalls } from '../memory/schema';

export function registerDebugRoutes(app: FastifyInstance) {
  app.get('/debug/db-stats', async () => {
    const convCount = db.select({ c: sql<number>`count(*)` }).from(conversations).get();
    const msgCount = db.select({ c: sql<number>`count(*)` }).from(messages).get();
    const memCount = db.select({ c: sql<number>`count(*)` }).from(typedMemories).get();
    const toolCount = db.select({ c: sql<number>`count(*)` }).from(toolCalls).get();

    const last = db
      .select()
      .from(messages)
      .orderBy(desc(messages.createdAt))
      .limit(1)
      .get();

    return {
      counts: {
        conversations: convCount?.c ?? 0,
        messages: msgCount?.c ?? 0,
        typed_memories: memCount?.c ?? 0,
        tool_calls: toolCount?.c ?? 0,
      },
      last_message: last
        ? {
            id: last.id,
            conversation_id: last.conversationId,
            role: last.role,
            content_preview: last.content.slice(0, 200),
            created_at: last.createdAt,
          }
        : null,
      database_path: process.env.DATABASE_PATH ?? '(default)',
      timestamp: new Date().toISOString(),
    };
  });
}
```
